# Share build settings while compiling every consumer with its own definitions.
function(cox_configure_map_server target)
	cox_target_defaults(${target})
	target_include_directories(${target} PRIVATE
		src ../Common ../Game/src ../DBServer/src)
	target_link_libraries(${target} PRIVATE
		AILib UtilitiesLib Lua::Lua PhysX::Loader PhysX::Cooking imm32)
	target_compile_definitions(${target} PRIVATE
		SERVER USEZLIB _CONSOLE _MBCS)
	target_compile_options(${target} PRIVATE
		$<$<CONFIG:Release,OptDebug>:/Gy;/Oi;/Zi>)
	target_link_options(${target} PRIVATE /LARGEADDRESSAWARE
		$<$<CONFIG:Release,OptDebug>:/DEBUG;/OPT:REF;/OPT:ICF>)
endfunction()

		${MapServer_RESOURCES} ${MapServer_IMAGES} ${MapServer_GENERATED})
	cox_configure_map_server(MapServer)
	cox_stage_physx(MapServer)
endif()
